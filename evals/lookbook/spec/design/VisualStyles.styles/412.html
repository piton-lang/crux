    Console · query

    Ready
    SELECT p.name AS project,
       count(*) AS open_overdue
  FROM tasks t
  JOIN projects p ON p.id = t.project_id
 WHERE t.status <> 'done'
   AND t.due_on < current_date
 GROUP BY p.name
 ORDER BY open_overdue DESC;
